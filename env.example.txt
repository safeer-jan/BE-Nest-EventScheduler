# ---------- App ----------
NODE_ENV=development
PORT=4000
API_PREFIX=api
# Comma-separated list of allowed browser origins (Vite dev server by default)
CORS_ORIGIN=http://localhost:5173
SWAGGER_ENABLED=true

# ---------- PostgreSQL (separate, empty database: scheduler_db) ----------
DB_HOST=localhost
DB_PORT=5433
DB_USERNAME=postgres
DB_PASSWORD=
DB_NAME=scheduler_db
DB_SSL=false
DB_LOGGING=false
# Run pending migrations automatically at boot (convenient for dev/test; prefer `npm run migration:run` in prod)
DB_MIGRATIONS_RUN=false

# ---------- Auth Service integration (VERIFY-ONLY) ----------
# This service never signs tokens. It verifies RS256 access tokens issued by your Auth Service.
# Copy the Auth Service's keys/public.pem to ./keys/public.pem (or paste the PEM into JWT_PUBLIC_KEY with \n escapes).
JWT_PUBLIC_KEY_PATH=keys/public.pem
# JWT_PUBLIC_KEY=
# MUST match the Auth Service's JWT_ISSUER / JWT_AUDIENCE
JWT_ISSUER=nest-auth-service
JWT_AUDIENCE=nest-auth-service-clients
# Role name (from the token's `roles` claim) that may see/manage every user's data
ADMIN_ROLE=admin

# ---------- Auth Service API (used by admins to list users for the event dropdown) ----------
# Base URL incl. prefix + version. The admin's own token is forwarded, so that admin needs the
# Auth Service permission `users:read` (the seeded admin role normally has it).
AUTH_SERVICE_URL=http://localhost:3000/api/v1
AUTH_SERVICE_TIMEOUT_MS=5000

# ---------- Optional: Redis token-revocation check ----------
# Set REDIS_HOST to the SAME Redis the Auth Service uses so logged-out tokens (blacklist:<jti>) are rejected here too.
# Leave empty to disable (tokens are then valid until they expire, default 15 min).
REDIS_HOST=
REDIS_PORT=6379
REDIS_PASSWORD=
# If Redis is unreachable: true = accept the (signature-valid) token, false = respond 503
REDIS_FAIL_OPEN=true

# ---------- Rate limiting (per IP) ----------
THROTTLE_TTL=60
THROTTLE_LIMIT=100
